// How It Works

Working Principle

An ultrasonic sensor measures distance continuously, and the live reading in centimeters is printed in large, easy-to-read text on a small SSD1306 OLED screen, updating twice a second.

// Source Code

Project code

C++ / Arduino 
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET    -1

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

#define TRIG_PIN 9
#define ECHO_PIN 10

void setup() {
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  Serial.begin(9600);
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println(F("OLED Not Found"));
    while (true);
  }

  display.clearDisplay();
  display.setTextSize(2); // Increase text size for numbers
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.display();
}

void loop() {
  long duration;
  float distance_cm;

  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  duration = pulseIn(ECHO_PIN, HIGH);
  distance_cm = duration * 0.034 / 2;

  display.clearDisplay();
  display.setCursor(0, 0);
  display.print(distance_cm, 1); // 1 digit after the decimal point
  display.print(" cm");
  display.display();

  delay(500);
}
